CREATE DATABASE samayGrupo;

USE samay;

CREATE TABLE user (
    user_id INT PRIMARY KEY AUTO_INCREMENT, 
    fullName VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    password_User VARCHAR(255) NOT NULL, 
    cellphone INT NOT NULL, 
    address VARCHAR(255) NOT NULL, 
    role_user VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE client (
    client_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    payment_method VARCHAR(50),
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES user(user_id)
);

CREATE TABLE order_user (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    clientId_fk INT NOT NULL, 
    fecha DATETIME,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (clientId_fk) REFERENCES client(client_id)
);

CREATE TABLE farmer (
    farmer_id INT PRIMARY KEY AUTO_INCREMENT,
    nombre_empresa_proveedor VARCHAR(100) NOT NULL,
    userId_fk INT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (userId_fk) REFERENCES user(user_id)
);

CREATE TABLE product (
    product_id INT PRIMARY KEY AUTO_INCREMENT,
    name_product VARCHAR(255) NOT NULL,
    stock INT NOT NULL,
    unit_price DECIMAL(10 , 2 ) NOT NULL,
    farmerId_fk INT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (farmerId_fk)
        REFERENCES farmer (farmer_id)
);

CREATE TABLE product_table (
    product_farmer_id INT PRIMARY KEY AUTO_INCREMENT,
    farmerId_fk INT NOT NULL,
    productId_fk INT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (farmerId_fk) REFERENCES farmer(farmer_id),
    FOREIGN KEY (productId_fk) REFERENCES product(product_id)
);
